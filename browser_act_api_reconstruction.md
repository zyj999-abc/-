## 使用示例

### 1. 反检测内容提取

```python
from browser_act_cli.stealth_extract import stealth_extract, ExtractionDiagnostics

# 基础提取
result = await stealth_extract(
    "https://protected-site.com",
    dynamic_proxy="US",
    content_type="markdown"
)

print(result.markdown)

# 获取诊断信息
diagnostics = ExtractionDiagnostics(result)
print(diagnostics.details())
```

### 2. 验证码绕过

```python
from browser_act_cli.session.captcha import CaptchaBypassService
from browser_act_cli.session.captcha.registry import get_solver

# 获取解决器
solver = get_solver("recaptcha_v2_token")
token = solver.solve()

# 或者使用服务
service = CaptchaBypassService()
service.bypass()
```

### 3. 浏览器会话管理

```python
from browser_act_cli.session.client import SessionClient
from browser_act_cli.session.protocol import Request

# 创建会话
client = SessionClient("ws://localhost:9222")

# 打开浏览器
request = Request.create("browser_open", browser_id="stealth-1", url="https://example.com")
response = await client.send(request)

# 导航
request = Request.create("navigate", url="https://example.com/page")
await client.send(request)

# 获取页面状态
request = Request.create("state")
state = await client.send(request)

# 点击元素
request = Request.create("click", index=3)
await client.send(request)
```

### 4. Cookie 管理

```python
# 导出 cookies
request = Request.create("cookies_export", file_path="/path/to/cookies.json")
await client.send(request)

# 导入 cookies
request = Request.create("cookies_import", file_path="/path/to/cookies.json")
await client.send(request)
```

### 5. 网络请求监控

```python
# 启动 HAR 捕获
request = Request.create("har_start")
await client.send(request)

# 触发页面操作...

# 停止 HAR 捕获并获取
request = Request.create("har_stop", output_path="/path/to/har.json")
har = await client.send(request)

# 获取单个请求详情
request = Request.create("network_detail", request_id="12345")
detail = await client.send(request)
```
